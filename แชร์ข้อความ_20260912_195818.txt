​local function flyToPlayer(targetPlr)
if not isRunning or not targetPlr.Character or not targetPlr.Character:FindFirstChild("HumanoidRootPart") then return end
​local targetHrp = targetPlr.Character.HumanoidRootPart
local targetCFrame = targetHrp.CFrame * CFrame.new(0, 2, -6)
local targetPos = targetCFrame.Position
​local distance = (freecamPos - targetPos).Magnitude
local duration = distance / gotoSpeed
local startTime = tick()
local startPos = freecamPos
​isTweening = true
​local lookCFrame = CFrame.lookAt(targetPos, targetHrp.Position)
local rx, ry, rz = lookCFrame:ToOrientation()
rotX = math.deg(ry)
rotY = math.deg(rx)
​while tick() - startTime < duration do
if not isRunning then break end
local alpha = (tick() - startTime) / duration
freecamPos = startPos:Lerp(targetPos, math.min(alpha, 1))
task.wait()
end
​if isRunning then
freecamPos = targetPos
end
isTweening = false
end
​local function findTargetPlayer()
local query = PlayerSearchInput.Text:lower()
if query == "" then return nil end
​for _, plr in pairs(Players:GetPlayers()) do
if plr ~= LocalPlayer then
if plr.Name:lower():find(query) or plr.DisplayName:lower():find(query) then
return plr
end
end
end
return nil
end

​function enableFreeCam()
local character = LocalPlayer.Character
if not character or not character:FindFirstChild("HumanoidRootPart") then return end
​local hrp = character.HumanoidRootPart
local humanoid = character:FindFirstChildOfClass("Humanoid")
local camera = workspace.CurrentCamera
​if humanoid then
humanoid.WalkSpeed = 0
humanoid.JumpPower = 0
end
​local startCFrame = hrp.CFrame * CFrame.new(0, 2, -5)
freecamPos = startCFrame.Position
​local rx, ry, rz = hrp.CFrame:ToOrientation()
rotX = math.deg(ry)
rotY = 0
​camera.CameraType = Enum.CameraType.Scriptable
​touchConnection = UserInputService.InputChanged:Connect(function(input, gameProcessed)
if input.UserInputType == Enum.UserInputType.Touch or input.UserInputType == Enum.UserInputType.MouseMovement then
if input.Position.X > (camera.ViewportSize.X * 0.25) then
local delta = input.Delta
rotX = rotX - (delta.X * SENSITIVITY)
rotY = math.clamp(rotY - (delta.Y * SENSITIVITY), -89, 89)
end
end
end)
​renderConnection = RunService.RenderStepped:Connect(function(dt)
if not isRunning then return end
local camera = workspace.CurrentCamera
local camRotation = CFrame.Angles(0, math.rad(rotX), 0) * CFrame.Angles(math.rad(rotY), 0, 0)
​if not isTweening and character and humanoid then
local moveVector = Vector3.new()
local rawDir = humanoid.MoveDirection
​if rawDir.Magnitude > 0 then
local camCFrame = CFrame.new(freecamPos) * camRotation
local currentCamLook = camera.CFrame.LookVector
local currentCamRight = camera.CFrame.RightVector
​local flatLook = Vector3.new(currentCamLook.X, 0, currentCamLook.Z)
if flatLook.Magnitude > 0 then flatLook = flatLook.Unit end
​local flatRight = Vector3.new(currentCamRight.X, 0, currentCamRight.Z)
if flatRight.Magnitude > 0 then flatRight = flatRight.Unit end
​local dotForward = rawDir:Dot(flatLook)
local dotRight = rawDir:Dot(flatRight)
​local moveDir = (camCFrame.LookVector * dotForward) + (camCFrame.RightVector * dotRight)
if moveDir.Magnitude > 0 then
moveVector = moveDir.Unit
end
end
​freecamPos = freecamPos + (moveVector * flySpeed * dt)
end
​camera.CFrame = CFrame.new(freecamPos) * camRotation
end)
end

humanoid.JumpPower = 50
if not isFollowing then
camera.CameraSubject = humanoid
end
end
end
​if not isFollowing then
camera.CameraType = Enum.CameraType.Custom
end
end
